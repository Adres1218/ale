Supabase Tabloları Eksik! (Kurulum Gerekli)

Mesajların ve aktif kullanıcıların Supabase veritabanına kaydedilebilmesi için SQL editöründe tabloları oluşturmanız gerekir. Lütfen aşağıdaki SQL kodunu kopyalayın, Supabase panelinizdeki SQL Editor alanına yapıştırın ve Run butonuna basın:

-- 1. Mesajlar tablosu
create table if not exists messages (
  id bigint generated always as identity primary key,
  text text not null,
  username text not null,
  avatar text,
  avatar_color text,
  city text,
  ip text,
  timestamp bigint,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Aktif kullanıcılar tablosu
create table if not exists active_users (
  id text primary key,
  username text not null,
  avatar text,
  avatar_color text,
  city text,
  ip text,
  last_active_at bigint not null
);
